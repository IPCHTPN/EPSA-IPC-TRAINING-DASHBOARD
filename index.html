⚙️ Setup Required — One Time Only (~3 minutes)

To enable real-time live data from your Google Sheet, you need to create a small Google Apps Script. This is free, takes 3 minutes, and only needs to be done once.

  1. Open script.google.com → click New Project
  2. Delete all existing code, then paste the code below
  3. Click Save (floppy disk icon)
  4. Click Deploy → New deployment
  5. Click the ⚙️ gear next to "Select type" → choose Web app
  6. Set Execute as: Me
  7. Set Who has access: Anyone
  8. Click Deploy → click Authorize access → choose your Google account → Allow
  9. Copy the Web app URL that appears
  10. Paste it below and click Connect
function doGet(e) { var SHEET_ID = '1q-L3OSZ6RJs0q1q0rxrJzZZ1j0aHD9qRJhoaMkaqYJw'; var ss = SpreadsheetApp.openById(SHEET_ID); var sheet = ss.getSheetByName('TEST MASTERLIST 2'); if (!sheet) sheet = ss.getSheets()[0]; var data = sheet.getDataRange().getValues(); var m1 = {}, m2 = {}; var inM1 = false, inM2 = false; for (var i = 0; i < data.length; i++) { var r = data[i]; var c0 = String(r[0] || '').trim().toUpperCase(); var c1 = String(r[1] || '').trim().toUpperCase(); // Detect M1 summary table header if (c0 === 'DEPARTMENT' && c1 === 'MODULE') { inM1=true; inM2=false; continue; } // Detect M2 summary table header if (c0 === 'DEPARTMENT' && (c1.indexOf('COMPLETED')>-1 || c1.indexOf('NUMERATOR')>-1)) { inM2=true; inM1=false; continue; } if (inM1) { if (!c0 || c0 === ':-:') { inM1=false; continue; } if (c1.indexOf('MODUL 1') > -1) { var sub = parseInt(String(r[2]||'').replace(/[^0-9]/g,'')) || 0; m1[c0] = sub; } } if (inM2) { if (!c0 || c0 === ':-:') { inM2=false; continue; } var sub2 = parseInt(String(r[1]||'').replace(/[^0-9]/g,'')); if (!isNaN(sub2)) m2[c0] = sub2; } } var result = { m1: m1, m2: m2, ts: new Date().toISOString() }; return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); }
