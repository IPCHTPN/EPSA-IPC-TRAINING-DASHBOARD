⚙️ One-time setup needed to enable live data

Because of browser security rules, the dashboard needs a small Google Apps Script to fetch your Sheet data. This takes about 3 minutes and only needs to be done once.

  1. Go to script.google.com and click New Project
  2. Delete all existing code and paste the code shown below
  3. Click Deploy → New deployment → Web app
  4. Set "Who has access" to Anyone
  5. Click Deploy → copy the Web App URL
  6. Paste the URL into the box below and click Connect
function doGet() { var sheet = SpreadsheetApp.openById('1q-L3OSZ6RJs0q1q0rxrJzZZ1j0aHD9qRJhoaMkaqYJw'); var output = {}; // Modul 1 summary (Pivot Table 4) var tab1 = sheet.getSheetByName('TEST MASTERLIST 2') || sheet.getSheets()[0]; var m1data = []; var m2data = []; var rows = tab1.getDataRange().getValues(); var inM1 = false, inM2 = false; for (var i = 0; i < rows.length; i++) { var r = rows[i]; var c0 = String(r[0]).trim().toUpperCase(); var c1 = String(r[1]).trim().toUpperCase(); // Detect M1 summary header if (c0 === 'DEPARTMENT' && c1 === 'MODULE') { inM1 = true; inM2 = false; continue; } // Detect M2 summary header if (c0 === 'DEPARTMENT' && c1.includes('COMPLETED')) { inM2 = true; inM1 = false; continue; } if (inM1 && c0 && c0 !== ':-:') { if (!r[1]) { inM1 = false; continue; } m1data.push({ dept: String(r[0]).trim(), submitted: parseInt(r[2])||0, denom: r[3]!==''?parseInt(r[3])||null:null }); } if (inM2 && c0 && c0 !== ':-:') { if (!r[1] && !r[2]) { inM2 = false; continue; } m2data.push({ dept: String(r[0]).trim(), submitted: parseInt(r[1])||0, denom: r[2]!==''?parseInt(r[2])||null:null }); } } output.m1 = m1data; output.m2 = m2data; output.updated = new Date().toISOString(); return ContentService.createTextOutput(JSON.stringify(output)) .setMimeType(ContentService.MimeType.JSON); }
